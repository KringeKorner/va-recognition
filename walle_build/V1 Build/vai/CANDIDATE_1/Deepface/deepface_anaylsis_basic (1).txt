# imports
import os
import cv2
import numpy as np
from deepface import DeepFace
from multiprocessing import Process
from multiprocessing.queues import Full, Empty

# ------------------ ENV / LIGHTWEIGHT SETTINGS ------------------
os.environ["TF_CPP_MIN_LOG_LEVEL"] = "3"

# ------------------ GLOBALS ------------------
VAI = None

# Fixed output order expected by CPM fusion
LABELS = ["angry", "disgust", "fear", "happy", "sad", "surprise", "neutral"]

# Lightweight inference settings
INFER_SIZE = (640, 360)   # resize before DeepFace
DETECTOR_BACKEND = "opencv"

# Cheap pre-check detector
FACE_CASCADE = cv2.CascadeClassifier(
    cv2.data.haarcascades + "haarcascade_frontalface_default.xml"
)

# ------------------ HELPER FUNCTIONS ------------------
def initialize(C2V, V2C, VAI_HALT, VAI_READY):
    global VAI
    print("STARTING VAI")
    VAI = Process(target=main, args=(C2V, V2C, VAI_HALT, VAI_READY), daemon=False)
    VAI.start()

def average(emotion_spread):
    """
    emotion_spread = list of [labels, scores]
    returns CPM-compatible dict
    """
    if not emotion_spread:
        return {
            "LABELS": LABELS,
            "AVG_CONF": [0.0] * len(LABELS)
        }

    scores_array = np.array([entry[1] for entry in emotion_spread], dtype=np.float32)
    avg_scores = np.round(np.mean(scores_array, axis=0), 3).tolist()

    return {
        "LABELS": LABELS,
        "AVG_CONF": avg_scores
    }

def clean_up(C2V, VAI_HALT):
    print("SHUTTING DOWN VAI")
    VAI_HALT.set()

    if VAI is not None:
        VAI.join(timeout=2.0)
        if VAI.is_alive():
            VAI.terminate()

    while not C2V.empty():
        try:
            _ = C2V.get_nowait()
        except Empty:
            break

    print("VAI TERMINATED")

def zero_result():
    return {
        "LABELS": LABELS,
        "AVG_CONF": [0.0] * len(LABELS)
    }

def normalize_emotion_dict(emotions):
    """
    Reorders DeepFace emotion dict to fixed CPM order.
    """
    return [float(emotions.get(lbl, 0.0)) for lbl in LABELS]

def quick_face_present(frame_bgr):
    """
    Very cheap OpenCV check before DeepFace.
    """
    gray = cv2.cvtColor(frame_bgr, cv2.COLOR_BGR2GRAY)
    faces = FACE_CASCADE.detectMultiScale(gray, scaleFactor=1.1, minNeighbors=5)
    return len(faces) > 0

# ------------------ MAIN ------------------
def main(C2V, V2C, VAI_HALT, VAI_READY):
    print("INITIATING VAI WARMUP SEQUENCE")

    # Warmup DeepFace emotion model once
    try:
        DeepFace.analyze(
            img_path=np.zeros((224, 224, 3), dtype=np.uint8),
            actions=["emotion"],
            enforce_detection=False,
            detector_backend=DETECTOR_BACKEND
        )
    except Exception:
        pass

    VAI_READY.set()
    print("VAI ONLINE\n")

    while not VAI_HALT.is_set():
        try:
            FRAMES = C2V.get(timeout=0.2)
        except Empty:
            continue

        if not FRAMES:
            try:
                V2C.put(zero_result(), timeout=0.2)
            except Full:
                pass
            continue

        # ------------------ CPU-LIGHT STRATEGY ------------------
        # Use only the most recent frame from CPM batch
        FRAME = FRAMES[-1]

        if FRAME is None:
            try:
                V2C.put(zero_result(), timeout=0.2)
            except Full:
                pass
            continue

        # Resize before any heavy processing
        try:
            FRAME_SMALL = cv2.resize(FRAME, INFER_SIZE)
        except Exception:
            FRAME_SMALL = FRAME

        # Cheap pre-check: if no face, skip DeepFace entirely
        if not quick_face_present(FRAME_SMALL):
            try:
                V2C.put(zero_result(), timeout=0.2)
            except Full:
                pass
            continue

        emotion_spread = []

        try:
            img_rgb = cv2.cvtColor(FRAME_SMALL, cv2.COLOR_BGR2RGB)

            results = DeepFace.analyze(
                img_path=img_rgb,
                actions=["emotion"],
                enforce_detection=False,
                detector_backend=DETECTOR_BACKEND
            )

            if isinstance(results, list) and len(results) > 0:
                face = results[0]
                emotions = face.get("emotion", {})
                scores = normalize_emotion_dict(emotions)
                emotion_spread.append([LABELS, scores])

            result = average(emotion_spread)

        except Exception:
            result = zero_result()

        try:
            V2C.put(result, timeout=0.2)
        except Full:
            continue
